import math

def minimax(depth, node, is_max, values, alpha, beta):

    if depth == 3:
        print("Leaf node:", node, "Value:", values[node])
        return values[node]

    if is_max:
        best = -math.inf
        print("\nMAX node:", node)

        for i in range(2):

            val = minimax(depth + 1, node * 2 + i,
                          False, values, alpha, beta)

            best = max(best, val)
            alpha = max(alpha, best)

            print("MAX Best:", best, "Alpha:", alpha)

            if beta <= alpha:
                print("Pruning at node:", node)
                break

        return best

    else:
        best = math.inf
        print("\nMIN node:", node)

        for i in range(2):

            val = minimax(depth + 1, node * 2 + i,
                          True, values, alpha, beta)

            best = min(best, val)
            beta = min(beta, best)

            print("MIN Best:", best, "Beta:", beta)

            if beta <= alpha:
                print("Pruning at node:", node)
                break

        return best


values = [3, 5, 6, 9, 1, 2, 0, -1]

print("Leaf values:", values)

result = minimax(0, 0, True, values, -math.inf, math.inf)

print("\nOptimal value:", result)
